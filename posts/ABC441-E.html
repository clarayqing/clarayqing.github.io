ABC441 pE

Editorial AtCoder

題目來源：ABC pE - A > B substring

題目輸入：給一個長度為 N 的字串 S 求有多少個子字串中的 A 比 B 多 字串 S 只有 A B C 三種字母


考慮子字串 [0, i] 時 要找出 [0, i-1] 的子字串有多少的 cnt_A-cnt_B 比現在的 cnt_A-cnt_B 少

利用樹狀陣列紀錄 cnt_A-cnt_B 的值 每次將 cnt_A-cnt_B 值增加一 就查找最小值到 cnt_A-cnt_B-1 的和 應為這些是在考慮這字母前的子字串的紀錄


初始值：

int const c=5e5;
int n;
int a[2*c+10]={0};

lowbit 函式：

int lowbit(int x){
    return x&-x;
}

點更新：

void update(int x,int k){
    while(x<=2*c+5)a[x]+=k,x+=lowbit(x);
}

區間求和

int query(int x){
    int t=0;
    while(x)t+=a[x],x-=lowbit(x);
    return t;
}

預處理 cnt_A-cnt_B 的值：

cin>>n;
string s;cin>>s;
vector<int> ps(n,0);
if(s[0]=='A')ps[0]=1;
if(s[0]=='B')ps[0]=-1;
for(int i=1;i<n;i++){
    if(s[i]=='A')ps[i]=ps[i-1]+1;
    else if(s[i]=='B')ps[i]=ps[i-1]-1;
    else ps[i]=ps[i-1];
}

答案計算：

long long ans=0;
update(c+1,1);
for(int i=0;i<n;i++){
    update(ps[i]+c+1,1);
    ans+=query(ps[i]+c);
}
cout<<ans;
